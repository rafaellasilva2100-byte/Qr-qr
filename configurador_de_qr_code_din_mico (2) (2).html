A planilha vinculada abaixo já atualiza em tempo real. Você também pode exportar o arquivo .CSV ou colar este Google Apps Script em Extensões > Apps Script na sua planilha do Google para receber os dados automaticamente: 

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), data.roomId, data.label, data.targetUrl]);
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
